${channel.name.slice(0, 2).toUpperCase()}
${channel.name}

`; return; } elements.messages.innerHTML = messages .map((message) => { const sender = message.user || {}; const avatar = getAvatarUrl(sender); const text = message.text ? `
${escapeHtml(message.text)}
` : ''; let mediaMarkup = ''; if (message.media_type === 'image' && message.media_url) { mediaMarkup = `
`; } else if (message.media_type === 'video' && message.media_url) { mediaMarkup = `
`; } else if (message.media_type === 'audio' && message.media_url) { mediaMarkup = `
`; } else if (message.media_type === 'file' && message.media_url) { mediaMarkup = `
Open file
`; } return `

const response = await apiFetch('/api/channels', { method: 'POST', body: JSON.stringify({ name, description }), headers: { 'Content-Type': 'application/json' }, }); state.channels.push(response); state.activeChannelId = response.id;
